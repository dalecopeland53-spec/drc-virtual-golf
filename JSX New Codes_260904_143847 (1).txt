export function GolfAppProvider({ children }) {
const [handicap, setHandicap] = useState(18);
const [currentHoleIndex, setCurrentHoleIndex] = useState(0);
const [scores, setScores] = useState({});
const [voiceLog, setVoiceLog] = useState("Caddie System Status: Ready for voice simulation...");
const [isSunlightMode, setIsSunlightMode] = useState(false);
const [useMetres, setUseMetres] = useState(false);

const activeHole = production18HoleCourseData[currentHoleIndex];

const nextHole = () => {
if (currentHoleIndex < production18HoleCourseData.length - 1) {
setCurrentHoleIndex(currentHoleIndex + 1);
}
};

const prevHole = () => {
if (currentHoleIndex > 0) {
setCurrentHoleIndex(currentHoleIndex - 1);
}
};

const updateScore = (score) => {
const numericScore = parseInt(score) || 0;
setScores(prev => ({ ...prev, [activeHole.hole]: numericScore }));

if (numericScore > 0) {
const phrases = [
Caddie Audio Feed: "Stroke logged successfully for Hole ${activeHole.hole}.",Caddie Audio Feed: "Score updated to ${numericScore}. Stay focused.",
Caddie Audio Feed: "Confirmed. Saving card parameters."
 ];
setVoiceLog(phrases[Math.floor(Math.random() * phrases.length)]);
}
};

const handleVoiceCommand = (text) => {
const lowerText = text.toLowerCase();
if (lowerText.includes("next")) {
nextHole();
setVoiceLog("Voice Executed: Advanced to next hole.");
} else if (lowerText.includes("back") || lowerText.includes("previous")) {
prevHole();
setVoiceLog("Voice Executed: Returned to previous hole.");
} else if (lowerText.includes("score")) {
const match = lowerText.match(/\d+/);
if (match) {
updateScore(match[0]);
}
} else {
setVoiceLog(Command '${text}' not recognised. Try 'next hole' or 'score 4'.`);
}
};

return (
<GolfAppContext.Provider value={{
handicap, setHandicap,
activeHole, currentHoleIndex, setCurrentHoleIndex,
scores, updateScore,
nextHole, prevHole,
voiceLog, setVoiceLog, handleVoiceCommand,
isSunlightMode, setIsSunlightMode,
useMetres, setUseMetres,
courseData: production18HoleCourseData
}}>
{children}
</GolfAppContext.Provider>
);
}

export default function App() {

}
